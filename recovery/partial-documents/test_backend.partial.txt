class DatasetTests(unittest.TestCase):
    def setUp(self):

        shutil.copytree(ROOT/'knowledge/curated',self.root,ignore=shutil.ignore_patterns('._*'))
        self.manifest=self.root/'manifest.json'
    def tearDown(self): self.temp.cleanup()

        (self.root/'extra.md').write_text('NEVER IMPORT THIS DOCUMENT')
        docs=load_dataset(self.manifest)
        self.assertEqual(len(docs),4)

        with self.assertRaises(DatasetError): load_dataset(self.manifest)

        p=self.root/'articles/mcele-course-launch-sts1-student.json'
